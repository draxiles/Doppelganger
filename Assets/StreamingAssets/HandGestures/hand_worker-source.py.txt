"""Offline MediaPipe hand and optional head pose inference. Owns no devices.

stdin: little-endian <4siiiiiQ> header, magic HND1, width, height,
clockwise rotation, vertical-mirror flag, RGB byte count, sequence; RGB24
pixels in Unity bottom-left row order. stdout: one JSON object per line.
Only the Unity owner starts/stops devices and decides when commands execute.
"""
import os
os.environ.setdefault("TF_CPP_MIN_LOG_LEVEL", "3")
os.environ.setdefault("GLOG_minloglevel", "2")
import sys
import json
import math
import struct
import time
from contextlib import ExitStack
from pathlib import Path
import numpy as np
import mediapipe as mp


def emit(value):
    sys.stdout.write(json.dumps(value, ensure_ascii=True, separators=(",", ":")) + "\n")
    sys.stdout.flush()


def read_exact(stream, count):
    chunks = bytearray()
    while len(chunks) < count:
        block = stream.read(count - len(chunks))
        if not block:
            return None
        chunks.extend(block)
    return chunks


def make_landmarker(model):
    return mp.tasks.vision.HandLandmarker.create_from_options(
        mp.tasks.vision.HandLandmarkerOptions(
            # Model bytes also support project directories with Cyrillic names.
            base_options=mp.tasks.BaseOptions(model_asset_buffer=Path(model).read_bytes()),
            running_mode=mp.tasks.vision.RunningMode.VIDEO,
            num_hands=2,
            min_hand_detection_confidence=0.65,
            min_hand_presence_confidence=0.65,
            min_tracking_confidence=0.6))


def make_face_landmarker(model):
    return mp.tasks.vision.FaceLandmarker.create_from_options(
        mp.tasks.vision.FaceLandmarkerOptions(
            base_options=mp.tasks.BaseOptions(model_asset_buffer=Path(model).read_bytes()),
            running_mode=mp.tasks.vision.RunningMode.VIDEO,
            num_faces=1,
            min_face_detection_confidence=0.6,
            min_face_presence_confidence=0.65,
            min_tracking_confidence=0.6,
            output_facial_transformation_matrixes=True))


def extract_head(result):
    if not result.facial_transformation_matrixes:
        return {"Tracked": False}
    # The canonical face points toward the camera along +Z. A turn toward the
    # player's right moves that forward vector left in the unmirrored image.
    x, y, z = result.facial_transformation_matrixes[0][:3, 2]
    if not all(math.isfinite(v) for v in (x, y, z)) or z < 0.2:
        return {"Tracked": False}
    return {"Tracked": True,
            "Yaw": round(-math.degrees(math.atan2(x, z)), 3),
            "Pitch": round(-math.degrees(math.atan2(y, math.hypot(x, z))), 3)}


def extract(result):
    hands = []
    for i, landmarks in enumerate(result.hand_landmarks):
        category = result.handedness[i][0]
        def points(items):
            return [{"x": round(p.x, 6), "y": round(p.y, 6), "z": round(p.z, 6)} for p in items]
        hands.append({"Side": category.category_name,
                      "Confidence": round(category.score, 6),
                      "Landmarks": points(landmarks),
                      "WorldLandmarks": points(result.hand_world_landmarks[i])})
    return hands


def main():
    root = Path(sys.executable).resolve().parent if getattr(sys, "frozen", False) else Path(__file__).resolve().parent
    model = root / "hand_landmarker.task"
    if len(sys.argv) > 1 and sys.argv[1] == "--self-test":
        from PIL import Image
        # Test images are explicit files, never physical devices.
        with make_landmarker(model) as detector:
            report = []
            for number, path in enumerate(sys.argv[2:]):
                rgb = np.array(Image.open(path).convert("RGB"))
                result = detector.detect_for_video(mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb), (number + 1) * 1000)
                report.append({"File": str(path), "Hands": extract(result)})
            emit({"Kind": "test", "Results": report})
        return
    with ExitStack() as stack:
        detector = stack.enter_context(make_landmarker(model))
        face_detector = stack.enter_context(make_face_landmarker(root / "face_landmarker.task")) if "--head" in sys.argv[1:] else None
        emit({"Kind": "ready", "Status": "MediaPipe / CPU / offline", "HeadTracking": face_detector is not None})
        stream = sys.stdin.buffer
        timestamp = 0
        frame_index = 0
        while True:
            header = read_exact(stream, 32)
            if header is None:
                return
            magic, width, height, rotation, mirror, length, sequence = struct.unpack("<4siiiiiQ", header)
            if magic != b"HND1" or not (16 <= width <= 1920 and 16 <= height <= 1080) or length != width * height * 3:
                raise ValueError("Invalid or oversized RGB frame")
            frame = read_exact(stream, length)
            if frame is None:
                return
            started = time.monotonic()
            rgb = np.frombuffer(frame, dtype=np.uint8).reshape(height, width, 3)
            rgb = np.flipud(rgb)  # Unity pixels start at bottom-left.
            if mirror:
                rgb = np.flipud(rgb)
            rgb = np.ascontiguousarray(np.rot90(rgb, -(rotation // 90) % 4))
            timestamp += max(1, int((time.monotonic() - started) * 1000))
            # Only differences matter to MediaPipe. Keep timestamps monotonic.
            timestamp = max(timestamp, int(time.monotonic() * 1000))
            image = mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb)
            response = {"Kind": "result", "Sequence": sequence}
            if face_detector is not None:
                response["Head"] = extract_head(face_detector.detect_for_video(image, timestamp))
            # Prefer a fresh view angle over repeated hand inference. A gesture already
            # needs a sustained pose; the Unity side keeps the last hand sample briefly.
            response["HandSample"] = face_detector is None or frame_index % 2 == 0
            if response["HandSample"]:
                response["Hands"] = extract(detector.detect_for_video(image, timestamp))
            frame_index += 1
            response["DurationMs"] = round((time.monotonic() - started) * 1000, 2)
            emit(response)


if __name__ == "__main__":
    try:
        main()
    except Exception as error:
        emit({"Kind": "error", "Status": type(error).__name__ + ": " + str(error)})
        sys.exit(1)
